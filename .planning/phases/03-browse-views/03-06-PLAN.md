    ```
    export async function deleteMeaning(meaningId: string): Promise<void> {
      await db.transaction('rw', db.wordFormMeanings, db.meanings, async () => {
        // Cascade: delete all wordFormMeanings junction rows referencing this meaning
        await db.wordFormMeanings.where('meaningId').equals(meaningId).delete();
        // Then delete the meaning record itself
        await db.meanings.delete(meaningId);
      });
    }
    ```

    ```
    export async function getWordFormsWithActiveMeaningCount() {
      return db.wordForms
        .toArray()
        .then(wordForms =>
          Promise.all(
            wordForms.map(async wf => {
              const activeMeaningCount = await db.wordFormMeanings
                .where('wordFormId').equals(wf.id)
                .toArray()
                .then(links =>
                  db.meanings
                    .where('id').anyOf(links.map(l => l.meaningId))
                    .filter(m => m.isActive === true)
                    .count()
                );
              return { ...wf, activeMeaningCount };
            })
          )
        );
    }
    ```
